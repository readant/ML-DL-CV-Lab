03三条结论（都能在 demo 输出里看到数字）

01

Wq 与 Wk 必须分开，否则注意力退化成「照镜子」

把 Q 与 K 绑定（用同一个矩阵）时，分数矩阵对称，每个词与自己最相似：91% 的词最强关注自己，对角线平均权重 0.498。换成两个独立矩阵后，这两个数字降到 9% 和 0.088。这正是每个头都要有独立 Wq、Wk 的原因 —— 不是重复劳动。

02

多头不是 h 倍参数，而是 h 份并行

因为 dk = d/h，四个投影矩阵各 d×d，总量恒为 4d²，与头数无关（断言 4 直接验证）。多头的代价是并行度与显存，不是参数量。所以「多头比单头强」不可能来自"参数更多"这个解释。

03

mask 必须加在 softmax 之前

先 softmax 再乘 0/1，被屏蔽位置的概率质量仍然计入了归一化分母，剩下有效位置的权重和 < 1，输出被系统性缩小。断言 3 用数字把这个损失量算了出来：正好等于被屏蔽位置原本占的概率质量。因果掩码下第一个词的权重必须是 1.000，demo 输出里可以看到。

attention_numpy.py 纯 NumPy 从零实现 · 177 行 
"""
attention_numpy.py —— 注意力 / 多头注意力 的纯 NumPy 实现（从零写，不调库）

设计原则
--------
1. 每个函数都标注张量形状，形状错了立刻报错，不靠猜。
2. 公式与实现一一对应，注释里给出对应书页码：
      d2l 第 10 章     p.399–430（《动手学深度学习》中文版）
      Bishop 12.1      p.374–384（Deep Learning: Foundations and Concepts, 2024）
      从零构建 LLM 中文版 p.64–97（第 3 章 编码注意力机制）
3. 只依赖 numpy。对拍用的 PyTorch 版在 attention_torch.py，两边权重可以互拷。

符号约定
--------
B  batch size
T  序列长度（query 侧）
S  序列长度（key/value 侧；自注意力时 S == T）
d  d_model，模型宽度
h  head 数
dk d_model // h，每个头的维度
"""

from __future__ import annotations

import numpy as np

__all__ = [
    "softmax",
    "scaled_dot_product_attention",
    "causal_mask",
    "MultiHeadAttention",
    "sinusoidal_positional_encoding",
]


def softmax(x: np.ndarray, axis: int = -1, mask: np.ndarray | None = None) -> np.ndarray:
    """数值稳定的 softmax。

    mask: 与 x 同形（可广播），True 表示「允许注意」，False 表示「屏蔽」。
          实现方式是把被屏蔽位置先置为 -inf 再归一化 —— 这是与「先 softmax 再乘 mask」
          的关键区别：后者会让被屏蔽位置的权重在归一化分母里仍然占份额。
    """
    if mask is not None:
        x = np.where(mask, x, -np.inf)
        # 整行全屏蔽时 softmax 会得到 nan；置 0 处理，避免污染下游
        all_masked = ~np.any(mask, axis=axis, keepdims=True)
        x = np.where(all_masked, 0.0, x)

    x_max = np.max(x, axis=axis, keepdims=True)
    x_max = np.where(np.isfinite(x_max), x_max, 0.0)
    e = np.exp(x - x_max)
    s = np.sum(e, axis=axis, keepdims=True)
    out = e / np.where(s == 0, 1.0, s)
    return out


def scaled_dot_product_attention(
    Q: np.ndarray,
    K: np.ndarray,
    V: np.ndarray,
    mask: np.ndarray | None = None,
) -> tuple[np.ndarray, np.ndarray]:
    """缩放点积注意力（d2l p.415 / 从零构建 LLM p.84）。

        Attention(Q, K, V) = softmax(Q Kᵀ / √dk) V

    为什么除以 √dk：q·k = Σ q_i k_i，若 q、k 各分量独立且方差为 1，则点积方差为 dk。
    不缩放时 dk 越大点积越极端，softmax 进入饱和区、梯度接近 0（Bishop 12.1.5）。

    参数
    ----
    Q : (..., T, dk)
    K : (..., S, dk)
    V : (..., S, dv)
    mask : (T, S) 或可广播到 (..., T, S)，True=允许注意

    返回
    ----
    out     : (..., T, dv)
    weights : (..., T, S)  每行和为 1（整行被屏蔽时为 0）
    """
    dk = Q.shape[-1]
    if K.shape[-1] != dk:
        raise ValueError(f"Q 与 K 的最后一维必须相同，得到 {Q.shape} 与 {K.shape}")

    scores = Q @ np.swapaxes(K, -1, -2) / np.sqrt(dk)  # (..., T, S)
    weights = softmax(scores, axis=-1, mask=mask)  # (..., T, S)
    out = weights @ V  # (..., T, dv)
    return out, weights


def causal_mask(T: int, S: int | None = None) -> np.ndarray:
    """因果掩码：位置 i 只能看到 j <= i（GPT 类解码器用，从零构建 LLM p.147）。"""
    S = T if S is None else S
    idx_q = np.arange(T)[:, None]
    idx_k = np.arange(S)[None, :]
    return idx_k <= idx_q  # (T, S)，下三角为 True


class MultiHeadAttention:
    """多头注意力（Bishop 12.1.6 p.382 / d2l 10.5 p.422）。

    与「把同一个注意力算 h 遍」的区别：每个头有**独立的投影矩阵**，
    因此能同时关注不同的表示子空间。因为 dk = d/h，总参数量仍约为 4d²，
    与单头基本相同 —— 多头的代价是并行度而不是参数量。

    参数量：W_q(d×d) + W_k(d×d) + W_v(d×d) + W_o(d×d) = 4d²
    """

    def __init__(self, d_model: int, num_heads: int, rng: np.random.Generator | None = None):
        if d_model % num_heads != 0:
            raise ValueError(f"d_model={d_model} 必须能被 num_heads={num_heads} 整除")
        self.d_model = d_model
        self.num_heads = num_heads
        self.dk = d_model // num_heads

        rng = rng or np.random.default_rng(0)
        # Xavier 风格缩放，避免初始 score 方差过大
        scale = 1.0 / np.sqrt(d_model)
        self.W_q = rng.normal(0, scale, (d_model, d_model))
        self.W_k = rng.normal(0, scale, (d_model, d_model))
        self.W_v = rng.normal(0, scale, (d_model, d_model))
        self.W_o = rng.normal(0, scale, (d_model, d_model))

    # ---- 形状变换 ----
    def _split_heads(self, x: np.ndarray) -> np.ndarray:
        """(B, T, d) -> (B, h, T, dk)"""
        B, T, _ = x.shape
        return x.reshape(B, T, self.num_heads, self.dk).transpose(0, 2, 1, 3)

    def _merge_heads(self, x: np.ndarray) -> np.ndarray:
        """(B, h, T, dk) -> (B, T, d)"""
        B, h, T, dk = x.shape
        return x.transpose(0, 2, 1, 3).reshape(B, T, h * dk)

    # ---- 前向 ----
    def forward(
        self,
        x_q: np.ndarray,
        x_kv: np.ndarray | None = None,
        mask: np.ndarray | None = None,
    ) -> tuple[np.ndarray, np.ndarray]:
        """x_q: (B, T, d)；x_kv: (B, S, d)，None 表示自注意力。

        返回 out (B, T, d) 与 weights (B, h, T, S)。
        """
        x_kv = x_q if x_kv is None else x_kv
        if x_q.ndim != 3 or x_kv.ndim != 3:
            raise ValueError("输入必须是 (B, T, d) 三维张量")

        Q = self._split_heads(x_q @ self.W_q)  # (B, h, T, dk)
        K = self._split_heads(x_kv @ self.W_k)  # (B, h, S, dk)
        V = self._split_heads(x_kv @ self.W_v)  # (B, h, S, dk)

        ctx, w = scaled_dot_product_attention(Q, K, V, mask)  # (B,h,T,dk), (B,h,T,S)
        out = self._merge_heads(ctx) @ self.W_o  # (B, T, d)
        return out, w

    __call__ = forward

    def num_parameters(self) -> int:
        return sum(p.size for p in (self.W_q, self.W_k, self.W_v, self.W_o))


def sinusoidal_positional_encoding(max_len: int, d_model: int) -> np.ndarray:
    """正弦位置编码（d2l 10.6）。

    必要性：自注意力本身对顺序**不敏感** —— 打乱词序，输出只是跟着重排，
    语义不变。位置编码把「位置」注入 token 表示，才让模型知道先后。
    """
    pe = np.zeros((max_len, d_model))
    pos = np.arange(max_len)[:, None]  # (max_len, 1)
    i = np.arange(0, d_model, 2)[None, :]  # (1, d/2)
    angle = pos / np.power(10000.0, i / d_model)
    pe[:, 0::2] = np.sin(angle)
    pe[:, 1::2] = np.cos(angle[:, : d_model // 2])
    return pe  # (max_len, d_model)
attention_torch.py PyTorch 可训练实现 · 158 行 
"""
attention_torch.py —— 注意力 / 多头注意力的 PyTorch 实现（nn.Module，可训练）

与 attention_numpy.py 的区别：
  · 这里带 autograd，能反传、能训练；
  · 参数排布按 PyTorch 习惯（把 W_q/W_k/W_v 合成一个 in_proj，方便一次矩阵乘），
    但提供 load_from_numpy() / to_numpy_params() 与 NumPy 版逐位对拍。

形状约定与 NumPy 版完全一致：输入 (B, T, d)，输出 (B, T, d)，权重 (B, h, T, S)。
"""

from __future__ import annotations

import math

import torch
import torch.nn as nn
import torch.nn.functional as F

__all__ = [
    "ScaledDotProductAttention",
    "MultiHeadAttention",
    "TransformerBlock",
    "causal_mask_bool",
]


def causal_mask_bool(T: int, S: int | None = None, device=None) -> torch.Tensor:
    """(T, S) 布尔掩码，True=允许注意（下三角）。与 numpy 版 causal_mask 等价。"""
    S = T if S is None else S
    return torch.ones(T, S, dtype=torch.bool, device=device).tril()


class ScaledDotProductAttention(nn.Module):
    """softmax(QKᵀ/√dk + mask) V

    注意 mask 的加法位置：**必须在 softmax 之前**加 -inf。
    若改成 softmax 之后再乘 0，被屏蔽位置的概率质量已经计入分母，
    剩下的有效位置权重之和不等于 1，输出会被系统性缩小。
    """

    def __init__(self, dropout: float = 0.0):
        super().__init__()
        self.dropout = nn.Dropout(dropout)

    def forward(
        self,
        q: torch.Tensor,
        k: torch.Tensor,
        v: torch.Tensor,
        mask: torch.Tensor | None = None,
    ) -> tuple[torch.Tensor, torch.Tensor]:
        dk = q.size(-1)
        scores = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(dk)  # (B,h,T,S)
        if mask is not None:
            scores = scores.masked_fill(~mask, float("-inf"))
        w = F.softmax(scores, dim=-1)
        w = self.dropout(w)
        return torch.matmul(w, v), w


class MultiHeadAttention(nn.Module):
    """多头注意力。参数量 4d²，与头数无关（d 固定时）。"""

    def __init__(self, d_model: int, num_heads: int, dropout: float = 0.0, bias: bool = True):
        super().__init__()
        if d_model % num_heads != 0:
            raise ValueError(f"d_model={d_model} 必须能被 num_heads={num_heads} 整除")
        self.d_model = d_model
        self.num_heads = num_heads
        self.dk = d_model // num_heads

        # 一次算出 [Q|K|V]，再切三块（PyTorch nn.MultiheadAttention 也是这么做的）
        self.in_proj = nn.Linear(d_model, 3 * d_model, bias=bias)
        self.out_proj = nn.Linear(d_model, d_model, bias=bias)
        self.attn = ScaledDotProductAttention(dropout)

    def _split(self, x: torch.Tensor) -> torch.Tensor:
        """(B,T,d) -> (B,h,T,dk)"""
        B, T, _ = x.shape
        return x.view(B, T, self.num_heads, self.dk).transpose(1, 2)

    def _merge(self, x: torch.Tensor) -> torch.Tensor:
        """(B,h,T,dk) -> (B,T,d)"""
        B, h, T, dk = x.shape
        return x.transpose(1, 2).contiguous().view(B, T, h * dk)

    def forward(
        self,
        x_q: torch.Tensor,
        x_kv: torch.Tensor | None = None,
        mask: torch.Tensor | None = None,
        is_causal: bool = False,
        need_weights: bool = True,
    ):
        x_kv = x_q if x_kv is None else x_kv
        B, T, _ = x_q.shape
        S = x_kv.size(1)
        d = self.d_model

        if x_kv is x_q:  # 自注意力：一次投影出 Q|K|V
            qkv = self.in_proj(x_q)  # (B,T,3d)
            q = self._split(qkv[..., :d])
            k = self._split(qkv[..., d : 2 * d])
            v = self._split(qkv[..., 2 * d :])
        else:  # 交叉注意力：Q 与 KV 分别投影
            q = self._split(self.in_proj(x_q)[..., :d])
            kv = self.in_proj(x_kv)  # (B,S,3d)
            k = self._split(kv[..., d : 2 * d])
            v = self._split(kv[..., 2 * d :])

        if is_causal:
            m = causal_mask_bool(T, S, device=x_q.device)
            mask = m if mask is None else (mask & m)

        ctx, w = self.attn(q, k, v, mask)  # (B,h,T,dk), (B,h,T,S)
        out = self.out_proj(self._merge(ctx))  # (B,T,d)
        return (out, w) if need_weights else out

    # ---- 与 NumPy 版对拍 ----
    def load_from_numpy(self, m) -> "MultiHeadAttention":
        """把 attention_numpy.MultiHeadAttention 的四个 d×d 矩阵拷进来。"""
        with torch.no_grad():
            W = [
                torch.as_tensor(m.W_q, dtype=torch.float32),
                torch.as_tensor(m.W_k, dtype=torch.float32),
                torch.as_tensor(m.W_v, dtype=torch.float32),
            ]
            self.in_proj.weight.copy_(torch.cat(W, dim=1).t())  # nn.Linear 存的是 (out,in)
            self.in_proj.bias.zero_()
            self.out_proj.weight.copy_(torch.as_tensor(m.W_o, dtype=torch.float32).t())
            self.out_proj.bias.zero_()
        return self

    def num_parameters(self) -> int:
        return sum(p.numel() for p in self.parameters())


class TransformerBlock(nn.Module):
    """一个最小 Transformer 块：LN → MHA → 残差 → LN → FFN → 残差。

    放这个类是为了说明多头注意力的**位置**：它只是两个子层之一，
    另一半是前馈网络（逐 token 的 MLP），两者都靠残差与 LayerNorm 串起来。
    """

    def __init__(self, d_model: int, num_heads: int, d_ff: int | None = None, dropout: float = 0.0):
        super().__init__()
        d_ff = d_ff or 4 * d_model
        self.ln1 = nn.LayerNorm(d_model)
        self.mha = MultiHeadAttention(d_model, num_heads, dropout)
        self.ln2 = nn.LayerNorm(d_model)
        self.ffn = nn.Sequential(nn.Linear(d_model, d_ff), nn.GELU(), nn.Dropout(dropout), nn.Linear(d_ff, d_model))

    def forward(self, x: torch.Tensor, is_causal: bool = False):
        a, w = self.mha(self.ln1(x), is_causal=is_causal)  # Pre-LN 结构
        x = x + a
        x = x + self.ffn(self.ln2(x))
        return x, w

    q = TOKENS.index("it")
    for h in range(4):
        row = w[0, h, q]
        top = TOKENS[int(np.argmax(row))]
        ent = float(-np.sum(row * np.log(row + 1e-12)))
        print(f"  head {h}: «it» 最关注 «{top}»（{row.max():.3f}）　分布熵 {ent:.2f} （上限 ln{T} = {np.log(T):.2f}）")
    print()
    print("  熵越接近上限越平均。随机初始化的头差异是噪声；")
    print("  训练后才会分化成「有的抓句法、有的抓共指」—— 见 Voita et al. 2019 的头消融实验。")


if __name__ == "__main__":
    main()

08边界：这份代码不做什么

不含训练。没有数据集、没有 loss、没有梯度更新。要跑训练去看《从零构建大语言模型》第 3–4 章（本地中文版 p.64 起），那份代码是完整的。

不含 FlashAttention / 稀疏注意力 / 线性注意力。只保留标准实现，便于和公式逐行对照。想了解这些优化方向，看《大规模语言模型》2.3.2（p.49）。

demo 里的词向量是随机的。不演示「it 指向 animal」这类语义现象 —— 那需要训练好的权重，手写不出来。凡是用随机初始化能验证的（归一化、掩码、参数量、Q=K 退化），才放进 demo。
